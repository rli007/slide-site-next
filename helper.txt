def shrink_map(m, width="70%", height="500px"):
    """
    Constrain a Folium map to the given box size while preserving interactivity.
    Call immediately after you instantiate a folium.Map.
    """
    # Give the <div> explicit dimensions
    m.width  = 1000
    m.height = 1000

    # Override Leaflet's 100 % page‐filling CSS
    style_fix = f"""
    <style>
        html, body {{height: auto !important; width: auto !important; margin: 0;}}
        .folium-map {{margin: 2rem auto;}}   /* center horizontally */
    </style>
    """
    m.get_root().html.add_child(folium.Element(style_fix))
    return m

def load_recurring_routes_from_csv(filename):
    routes = []
    try:

            for row in reader:

        print(f"Error reading CSV file: {e}")

def get_distance_between_locations(gmaps, loc1, loc2):
    try:
        result = gmaps.distance_matrix(loc1, loc2, mode="driving")  # type: ignore
        if result['rows'][0]['elements'][0]['status'] == 'OK':
            return result['rows'][0]['elements'][0]['distance']['value']
        return float('inf')
    except Exception as e:
        print(f"Error getting distance between {loc1} and {loc2}: {e}")
        return float('inf')

def find_top_drivers_for_packages(gmaps, routes, packages, top_n=2):
    assignments = []
    
    for package_id, pickup, delivery, package_type, priority in packages:
        driver_scores = []
        
        for driver, origin, destination, days in routes:
            pickup_to_origin = get_distance_between_locations(gmaps, pickup, origin)
            destination_to_delivery = get_distance_between_locations(gmaps, destination, delivery)
            total_detour = pickup_to_origin + destination_to_delivery
            
            priority_multiplier = 1.0
            if priority == "High Priority":
                priority_multiplier = 0.7
            elif priority == "Low Priority":
                priority_multiplier = 1.3
            
            score = total_detour * priority_multiplier
            
            driver_scores.append({
                'driver': (driver, origin, destination, days, pickup_to_origin, destination_to_delivery),
                'score': score
            })
        
        top_drivers = sorted(driver_scores, key=lambda x: x['score'])[:top_n]
        
        assignments.append({
            'package': (package_id, pickup, delivery, package_type, priority),
            'top_drivers': top_drivers
        })
    
    return assignments

def extra_detour(origin, stop1, stop2, base_dist):
    """Distance origin→stop1→stop2 minus the normal origin→stop2 commute."""
    return get_distance_between_locations(gmaps, origin, stop1) + \
           get_distance_between_locations(gmaps, stop1, stop2) - base_dist


def best_two_driver_pair(gmaps, routes, pickup, delivery, priority):
    """
    Return (best_pair_info, best_score).
    best_pair_info = (D1_tuple, D2_tuple, transfer_latlng)
    """
    best_pair, best_score = None, float('inf')
    print("hi")
    # Pre-cache baseline commute distances so we don't hammer the API
    route_base = {}
    for drv, o, d, *_ in routes:
        route_base[(o, d)] = get_distance_between_locations(gmaps, o, d)

    for d1 in routes:
        drv1, o1, dest1, days1 = d1
        base1 = route_base[(o1, dest1)]

        # Driver-1 detour parts
        d1_extra = get_distance_between_locations(gmaps, o1, pickup)
        d1_extra += extra_detour(pickup, dest1, dest1, 0)  # pickup→dest1 minus 0

        # Transfer point is dest1
        transfer = dest1

        for d2 in routes:
            if d2 == d1:
                continue
            drv2, o2, dest2, days2 = d2
            base2 = route_base[(o2, dest2)]

            # Driver-2 detour parts
            d2_extra = extra_detour(o2, transfer, dest2, base2)  # origin2 via transfer
            d2_extra += get_distance_between_locations(gmaps, transfer, delivery)
            d2_extra += extra_detour(delivery, dest2, dest2, 0)  # delivery→dest2 minus 0

            total_extra = d1_extra + d2_extra

            # Priority weighting
            mult = 0.7 if priority == "High Priority" else 1.3 if priority == "Low Priority" else 1.0
            score = total_extra * mult

            if score < best_score:
                best_score = score
                best_pair = (d1, d2, transfer)

    return best_pair, best_score

def save_solo_map(pkg_info, solo_info):
    package_id, pickup, delivery, _, _ = pkg_info
    (drv, origin, dest, days), score, _ = solo_info
    map_mode = "solo"

    centre = gmaps.geocode(origin)[0]['geometry']['location']
    m = folium.Map(location=[centre["lat"], centre["lng"]-0.1], zoom_control=False, height=1000, width=800,zoom_start=10.5)

    # commute
    dir_commute = gmaps.directions(origin, dest, mode="driving")[0]
    coords = polyline.decode(dir_commute["overview_polyline"]["points"])
    folium.PolyLine(coords, color="cadetblue", weight=4, opacity=0.5,
                    tooltip=f"{drv} commute").add_to(m)

    # detour
    dir_detour = gmaps.directions(origin, dest, mode="driving",
                                  waypoints=[pickup, delivery])[0]
    dcoords = polyline.decode(dir_detour["overview_polyline"]["points"])
    folium.PolyLine(dcoords, color="black", weight=5, opacity=0.8,
                    dash_array="10,5",
                    tooltip=f"Detour route for {drv}").add_to(m)

    # markers
    for label, loc, col in [("Origin", origin, 'green'),
                            ("Pickup", pickup, 'blue'),
                            ("Delivery", delivery, 'orange'),
                            ("Destination", dest, 'red')]:
        g = gmaps.geocode(loc)[0]['geometry']['location']
        folium.Marker([g['lat'], g['lng']],
                      icon=folium.Icon(color=col),
                      popup=f"{label}: {loc}").add_to(m)

    name = f"package_{package_id.replace(' ', '_')}_solo.html"
    m.save(name);  print("Saved", name)

def build_split_page(map_file: str,

    Wrap an existing Folium map HTML in a two-column layout.

    - `map_file`      = path to the Folium file you just saved

def minutes_between(gmaps, a, b, default_kmh=45):
    """
    Try to get Google’s real drive-time in minutes.
    Fallback: distance / default_kmh.
    """
    try:
        result = gmaps.distance_matrix(a, b, mode="driving")["rows"][0]["elements"][0]
        if result["status"] == "OK":
            return round(result["duration"]["value"] / 60)   # seconds ➜ minutes
    except Exception:
        pass
    # fallback – use distance we already fetched:
    meters = get_distance_between_locations(gmaps, a, b)
    return round((meters/1000) / default_kmh * 60)
